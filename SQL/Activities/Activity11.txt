SELECT customer_id, customer_name 
FROM customers a
WHERE 1 < (
    -- Subquery counts number of orders per customer
    SELECT COUNT(*) 
    FROM orders b 
    WHERE a.customer_id = b.customer_id
)

UNION

SELECT salesman_id, salesman_name 
FROM salesman a
WHERE 1 < (
    -- Subquery counts number of orders per salesman
    SELECT COUNT(*) 
    FROM orders b 
    WHERE a.salesman_id = b.salesman_id
)

ORDER BY customer_name;


SELECT 
    s.salesman_id, 
    s.salesman_name, 
    o.order_no, 
    'Highest on' AS status,   
    o.order_date, 
    o.purchase_amount
FROM salesman s, orders o
WHERE s.salesman_id = o.salesman_id
AND o.purchase_amount = (
    SELECT MAX(purchase_amount) 
    FROM orders c 
    WHERE c.order_date = o.order_date
)

UNION

SELECT 
    s.salesman_id, 
    s.salesman_name, 
    o.order_no, 
    'Lowest on' AS status,   
    o.order_date, 
    o.purchase_amount
FROM salesman s, orders o
WHERE s.salesman_id = o.salesman_id
AND o.purchase_amount = (
    -- Subquery finds min purchase for that date
    SELECT MIN(purchase_amount) 
    FROM orders c 
    WHERE c.order_date = o.order_date
)

ORDER BY order_date;
